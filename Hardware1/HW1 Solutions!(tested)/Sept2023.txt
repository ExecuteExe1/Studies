module MealyFSM(input x,input rst,input clk,output reg y);
  parameter S0=2'b00;
  parameter S1=2'b01;
  parameter S2=2'b10;
  parameter S3=2'b11;
  reg [1:0] current_state,next_state;
  always@(posedge clk or negedge rst)begin
    if(rst)begin
      current_state<=S0;
    end else begin
       current_state<=next_state;
  end
  end
  always@(*)begin
   next_state=current_state;
       y=1'b0;
    case(current_state)
      S0:begin 
        case(x)
          2'b00:begin next_state=S1; y=1;end
          2'b01:next_state=S2;
           default next_state=S0;
          endcase
        end
      S1:begin
        case(x) 
          2'b00:next_state=S2;
          2'b01:begin next_state=S3;y=1; end
          default next_state=S1;
        endcase
       end
      S2:begin
        case(x)
          2'b00:next_state=S1;
          2'b01:begin next_state=S3;y=1; end
        default next_state=S2;
        endcase
       end
      S3:begin
        case(x)
          2'b00:begin next_state=S2;y=1; end
         2'b01:next_state=S0;
         default next_state=S0;
        endcase
       end
     default next_state=S0;
    endcase
  end
endmodule
