module mux2to1(sel, i1, i0, f);
    input i0, i1, sel;
    output f;
    wire nsel, w1, w2;

    not(nsel, sel);

    and(w1, i0, nsel);
    and(w2, i1, sel);
    or(f, w1, w2);
endmodule

module XorTest(input a,input b,input c,output z);
 wire y;
  mux2to1 Mux1(a,b,~b,y); 
  mux2to1 Mux2(y,c,~c,z);

endmodule

module Mux4to1 #(
    parameter width = 8
)(
    input  [width-1:0] a,
    input  [width-1:0] b,
  input  [width-1:0] c,
  input  [width-1:0] d,
  input       [1:0]   sel,
    output reg [width-1:0] z
);
  wire [width-1:0] w1,w2;
  Mux2to1 Mux1(a,b,sel[0],w1);
  Mux2to1 Mux2(c,d,sel[0],w2);
  Mux2to1 Mux3(w1,w2,sel[1],z);

endmodule
